57 double compute(double error) {
58 integral += error;
59 if (integralCap > 0.0) {
60 if (integral > integralCap) integral = integralCap;
61 if (integral < -integralCap) integral = -integralCap;

63 double deriv = seeded_err ? (error - prev_error) : 0.0;

86 seeded_meas = true;
